cmake_minimum_required(VERSION 3.16)
project(ArkForgeResources LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../_vendor" ABSOLUTE)
set(RESTOOL "${VENDOR}/developtools_global_resource_tool")
set(PNG_SHARED OFF CACHE BOOL "" FORCE)
set(PNG_STATIC ON CACHE BOOL "" FORCE)
set(PNG_TESTS OFF CACHE BOOL "" FORCE)
set(PNG_TOOLS OFF CACHE BOOL "" FORCE)
set(PNG_HARDWARE_OPTIMIZATIONS OFF CACHE BOOL "" FORCE)
set(ZLIB_INCLUDE_DIR "${VENDOR}/zlib" CACHE PATH "" FORCE)
set(ZLIB_LIBRARY "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-ohos/libz.so" CACHE FILEPATH "" FORCE)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/png")
configure_file("${VENDOR}/libpng/pnglibconf.h.prebuilt" "${CMAKE_CURRENT_BINARY_DIR}/png/pnglibconf.h" COPYONLY)
file(GLOB PNG_SOURCES "${VENDOR}/libpng/png*.c")
list(REMOVE_ITEM PNG_SOURCES "${VENDOR}/libpng/pngtest.c")
add_library(png_static STATIC ${PNG_SOURCES})
target_include_directories(png_static PUBLIC "${VENDOR}/libpng" "${CMAKE_CURRENT_BINARY_DIR}/png")
target_compile_definitions(png_static PRIVATE PNG_ARM_NEON_OPT=0)
target_link_libraries(png_static PUBLIC z)
file(GLOB SECUREC "${VENDOR}/securec/src/*.c")
add_library(arkforge_securec STATIC ${SECUREC})
target_include_directories(arkforge_securec PUBLIC "${VENDOR}/securec/include")
add_library(arkforge_cjson STATIC "${VENDOR}/cJSON/cJSON.c")
target_include_directories(arkforge_cjson PUBLIC "${VENDOR}/cJSON")
file(GLOB_RECURSE RESOURCE_SOURCES "${RESTOOL}/src/*.cpp")
list(REMOVE_ITEM RESOURCE_SOURCES "${RESTOOL}/src/restool.cpp")
add_library(arkforge_resource_core SHARED ${RESOURCE_SOURCES} core.cpp
  "${VENDOR}/zlib/contrib/minizip/unzip.c" "${VENDOR}/zlib/contrib/minizip/ioapi.c")
target_include_directories(arkforge_resource_core PRIVATE "${VENDOR}/zlib/contrib/minizip")
target_include_directories(arkforge_resource_core PRIVATE "${RESTOOL}/include" "${VENDOR}/libpng" "${CMAKE_CURRENT_BINARY_DIR}/png")
target_compile_definitions(arkforge_resource_core PRIVATE __LINUX__ unzLocateFile2=unzLocateFile)
target_compile_options(arkforge_resource_core PRIVATE -O2 -fvisibility=hidden)
target_link_libraries(arkforge_resource_core PRIVATE arkforge_securec arkforge_cjson png_static z)
add_library(arkforge_resources SHARED bridge.cpp workbench_server.cpp huawei_callback.cpp "${VENDOR}/zlib/contrib/minizip/unzip.c" "${VENDOR}/zlib/contrib/minizip/ioapi.c")
target_include_directories(arkforge_resources PRIVATE "${VENDOR}/zlib/contrib/minizip")
target_link_libraries(arkforge_resources PRIVATE libace_napi.z.so dl z arkforge_cjson)
