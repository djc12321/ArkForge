cmake_minimum_required(VERSION 3.20)
project(ArkForgeSigner C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT SIGNER_SOURCE)
    set(SIGNER_SOURCE "${VENDOR}/developtools_hapsigner/hapsigntool_cpp")
endif()
file(GLOB_RECURSE SIGNER_FILES CONFIGURE_DEPENDS "${SIGNER_SOURCE}/*.cpp")
list(FILTER SIGNER_FILES EXCLUDE REGEX "/main.cpp$")
file(GLOB_RECURSE ALL_HEADERS "${SIGNER_SOURCE}/*.h")
set(SIGNER_INCLUDES "${SIGNER_SOURCE}")
foreach(header ${ALL_HEADERS})
    get_filename_component(directory "${header}" DIRECTORY)
    list(APPEND SIGNER_INCLUDES "${directory}")
endforeach()
list(REMOVE_DUPLICATES SIGNER_INCLUDES)
file(GLOB SECUREC_FILES "${SECUREC_ROOT}/src/securec/*.c")
if(NOT SECUREC_FILES)
    file(GLOB SECUREC_FILES "${SECUREC_ROOT}/src/*.c")
endif()
add_library(securec STATIC ${SECUREC_FILES})
target_include_directories(securec PUBLIC "${SECUREC_ROOT}/include")
add_library(cjson STATIC "${VENDOR}/cJSON/cJSON.c")
target_include_directories(cjson PUBLIC "${VENDOR}/cJSON")
set(ZLIB_SOURCE "${VENDOR}/zlib")
add_library(minizip STATIC
    "${ZLIB_SOURCE}/contrib/minizip/unzip.c"
    "${ZLIB_SOURCE}/contrib/minizip/zip.c"
    "${ZLIB_SOURCE}/contrib/minizip/ioapi.c")
target_include_directories(minizip PUBLIC "${ZLIB_SOURCE}" "${ZLIB_SOURCE}/contrib/minizip")
target_compile_definitions(minizip PRIVATE USE_FILE32API)
add_library(signtool SHARED ../signer_bridge.cpp ${SIGNER_FILES})
target_include_directories(signtool PRIVATE ${SIGNER_INCLUDES} "${OPENSSL_ROOT}/include")
target_compile_options(signtool PRIVATE -fno-rtti -fvisibility=hidden -Wno-deprecated-declarations -Wno-c++11-narrowing)
target_link_options(signtool PRIVATE -Wl,--exclude-libs,ALL -Wl,--no-undefined)
target_link_libraries(signtool PRIVATE
    "${OPENSSL_ROOT}/lib/libssl.a" "${OPENSSL_ROOT}/lib/libcrypto.a"
    securec cjson minizip z ace_napi.z hilog_ndk.z)
